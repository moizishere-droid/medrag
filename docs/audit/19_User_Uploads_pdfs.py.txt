# CELL 1
# Run from either the repository root or the notebooks directory.
from pathlib import Path
import os
import sys
PROJECT_ROOT = next((p for p in [Path.cwd(), *Path.cwd().parents]
                     if (p / "backend" / "src" / "medrag").is_dir()), None)
if PROJECT_ROOT is None:
    raise RuntimeError("Start this notebook inside the MedRAG repository")
sys.path.insert(0, str(PROJECT_ROOT / "backend" / "src"))
sys.path.insert(0, str(PROJECT_ROOT / "backend"))
os.chdir(PROJECT_ROOT / "notebooks")
from config.settings import settings


# CELL 2
import uuid
from qdrant_client import QdrantClient, models as qm
from openai import OpenAI
from medrag.ingestion.user_upload import validate_upload_size, extract_text_from_pdf, chunk_user_upload, embed_and_upsert_upload_chunks
from medrag.retrieval.hybrid_search import build_user_filter
client = QdrantClient(url=settings.qdrant_url or "http://localhost:6333")
openai_client = OpenAI(api_key=settings.openai_api_key)
collection = "medrag_notebook19_" + uuid.uuid4().hex
client.create_collection(collection, vectors_config={"dense": qm.VectorParams(size=1536, distance=qm.Distance.COSINE)}, sparse_vectors_config={"sparse": qm.SparseVectorParams(modifier=qm.Modifier.IDF)})


# CELL 3
pdf_path = next((PROJECT_ROOT / "data").rglob("*.pdf"))
pdf_bytes = pdf_path.read_bytes()
validate_upload_size(pdf_bytes)
text = extract_text_from_pdf(pdf_bytes)
session_id = str(uuid.uuid4())
chunks = chunk_user_upload(text, user_id=session_id, session_id=session_id, document_id=str(uuid.uuid4()), filename=pdf_path.name)
count = embed_and_upsert_upload_chunks(chunks, client, openai_client, collection_name=collection)
assert client.count(collection, exact=True).count == count
assert client.count(collection, count_filter=build_user_filter(session_id), exact=True).count == count
assert client.count(collection, count_filter=build_user_filter(str(uuid.uuid4())), exact=True).count == 0
assert client.count(collection, count_filter=build_user_filter(None), exact=True).count == 0
print("Verified session isolation for", count, "uploaded chunks")


# CELL 4
# Cleanup is safe: this name was generated exclusively for this notebook run.
client.delete_collection(collection)
client.close()
openai_client.close()
